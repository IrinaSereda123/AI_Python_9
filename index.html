Работа 1

Создание базового помощника

Базовый код

import ollama

def ask_model(prompt):
    response = ollama.chat(
        model="qwen2:0.5b",
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    return response["message"]["content"]

if __name__ == "__main__":
    print("Локальный AI-помощник. Введите 'выход' для завершения")
    while True:
        user_input = input("\nВы: ")
        if user_input.lower() in ("выход", "exit"):
            break
        response = ask_model(user_input)
        print(f"AI: {response}")
Работа 2

Контекстный чат-бот

Цель: создать помощника с сохранением контекста диалога.

import ollama

class ContextChat:
    def __init__(self):
        self.messages = []
    
    def chat(self, user_input):
        self.messages.append({"role": "user", "content": user_input})
        response = ollama.chat(
            model="qwen2:0.5b",
            messages=self.messages
        )
        bot_response = response["message"]["content"]
        self.messages.append({"role": "assistant", "content": bot_response})
        return bot_response

if __name__ == "__main__":
    chat = ContextChat()
    print("Контекстный AI-помощник. Введите 'выход' для завершения")
    while True:
        user_input = input("\nВы: ")
        if user_input.lower() in ("выход", "exit"):
            break
        print(f"AI: {chat.chat(user_input)}")
Работа 3

Мультиагентная система

Цель: создать систему с несколькими агентами, решающими задачу совместно.

import ollama

class Agent:
    def __init__(self, role):
        self.role = role
        self.messages = []
    
    def think(self, prompt):
        self.messages.append({"role": "user", "content": f"Ты {self.role}. {prompt}"})
        response = ollama.chat(
            model="qwen2:0.5b",
            messages=self.messages
        )
        self.messages.append({"role": "assistant", "content": response["message"]["content"]})
        return response["message"]["content"]

class MultiAgentSystem:
    def __init__(self):
        self.analyst = Agent("аналитик")
        self.coder = Agent("программист")
        self.reviewer = Agent("рецензент")
    
    def solve_problem(self, problem):
        # Этап 1: Анализ проблемы
        analysis = self.analyst.think(problem)
        
         # Этап 2: Разработка решения
        code_solution = self.coder.think(f"На основе анализа создай код решения: {analysis}")
        
        # Этап 3: Проверка решения
        review = self.reviewer.think(f"Проанализируй код и предложи улучшения: {code_solution}")
        
        return {
            "анализ": analysis,
            "решение": code_solution,
            "ревью": review
        }

if __name__ == "__main__":
    print("Мультиагентная система. Введите 'выход' для завершения")
    system = MultiAgentSystem()
    
    while True:
        user_input = input("\nВведите задачу: ")
        if user_input.lower() in ("выход", "exit"):
            break
            
        result = system.solve_problem(user_input)
        print("\nАнализ проблемы:")
        print(result["анализ"])
        print("\nПредложенное решение:")
        print(result["решение"])
        print("\nРекомендации по улучшению:")
        print(result["ревью"])
